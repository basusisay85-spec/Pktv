🧩 One-file version (bot.py)

"""
Telegram bot: domain -> IP / ASN / hosting provider / geo.

Install:
    pip install aiogram aiohttp dnspython

Run:
    1. Put your @BotFather token in BOT_TOKEN below.
    2. python bot.py
    3. In Telegram: /start, then send "google.com"

(Optional) ipinfo token for higher rate limits:
    set IPINFO_TOKEN = "your_token"   # https://ipinfo.io/signup
"""

import asyncio
import socket
import aiohttp
import dns.resolver
from aiogram import Bot, Dispatcher, types
from aiogram.filters import Command

# ------------------------------------------------------------------
# CONFIG
# ------------------------------------------------------------------
BOT_TOKEN    = "YOUR_BOT_TOKEN_HERE"   # <-- @BotFather token
IPINFO_TOKEN = ""                      # <-- optional

# ------------------------------------------------------------------
# GLOBALS
# ------------------------------------------------------------------
bot = Bot(BOT_TOKEN)
dp  = Dispatcher()
session: aiohttp.ClientSession | None = None


# ------------------------------------------------------------------
# DNS RESOLUTION
# ------------------------------------------------------------------
async def resolve_ips(domain: str) -> list[str]:
    """Return all IPv4/IPv6 addresses for a domain (system resolver + DNS A/AAAA)."""
    ips: set[str] = set()
    loop = asyncio.get_running_loop()

    # System resolver
    try:
        infos = await loop.run_in_executor(None, socket.getaddrinfo, domain, None)
        for info in infos:
            ips.add(info[4][0])
    except Exception:
        pass

    # Explicit A / AAAA queries
    for rtype in ("A", "AAAA"):
        try:
            answers = await loop.run_in_executor(
                None,
                lambda rt=rtype: dns.resolver.resolve(domain, rt, lifetime=5),
            )
            for r in answers:
                ips.add(r.to_text())
        except Exception:
            pass

    return sorted(ips)


# ------------------------------------------------------------------
# IP INFO (ASN / provider / geo)
# ------------------------------------------------------------------
async def ip_info(ip: str) -> dict:
    url = f"https://ipinfo.io/{ip}/json"
    if IPINFO_TOKEN:
        url += f"?token={IPINFO_TOKEN}"
    try:
        timeout = aiohttp.ClientTimeout(total=10)
        async with session.get(url, timeout=timeout) as r:
            if r.status != 200:
                return {"error": f"HTTP {r.status}"}
            return await r.json()
    except Exception as e:
        return {"error": str(e)}


# ------------------------------------------------------------------
# FORMATTING
# ------------------------------------------------------------------
def _esc(s: str) -> str:
    """Escape Telegram Markdown special characters in dynamic strings."""
    for ch in ("_", "*", "[", "]", "`"):
        s = s.replace(ch, f"\\{ch}")
    return s


def format_ip_block(ip: str, data: dict) -> str:
    if "error" in data:
        return f"🌐 `{ip}` — request error: {data['error']}"

    org      = data.get("org", "—")
    city     = data.get("city", "—")
    region   = data.get("region", "—")
    country  = data.get("country", "—")
    hostname = data.get("hostname", "—")
    loc      = data.get("loc", "—")

    asn, provider = "—", org
    if org and org.startswith("AS"):
        parts = org.split(" ", 1)
        asn = parts[0]
        provider = parts[1] if len(parts) > 1 else "—"

    return (
        f"🌐 `{ip}`\n"
        f"   • ASN: {_esc(asn)}\n"
        f"   • Hosting/Provider: {_esc(provider)}\n"
        f"   • Country: {_esc(country)}, region: {_esc(region)}, city: {_esc(city)}\n"
        f"   • Coordinates: {_esc(loc)}\n"
        f"   • Reverse hostname: {_esc(hostname)}\n"
    )


# ------------------------------------------------------------------
# HANDLERS
# ------------------------------------------------------------------
@dp.message(Command("start"))
async def cmd_start(msg: types.Message):
    await msg.answer(
        "👋 Send a domain (e.g. `google.com` or `https://github.com`) — "
        "I'll return its IP, ASN, hosting provider and geolocation.",
        parse_mode="Markdown",
    )


@dp.message()
async def handle_domain(msg: types.Message):
    text = (msg.text or "").strip()
    if not text:
        return

    # Normalize: strip scheme, path, port, trailing dots
    domain = text.lower()
    for prefix in ("http://", "https://"):
        if domain.startswith(prefix):
            domain = domain[len(prefix):]
    domain = domain.split("/")[0].split(":")[0].strip().strip(".")

    if "." not in domain or " " in domain:
        await msg.answer(
            "❌ That doesn't look like a domain. Example: `example.com`",
            parse_mode="Markdown",
        )
        return

    wait = await msg.answer(f"🔎 Checking `{domain}` ...", parse_mode="Markdown")

    ips = await resolve_ips(domain)
    if not ips:
        await wait.edit_text(
            f"❌ Failed to resolve `{domain}`.", parse_mode="Markdown"
        )
        return

    blocks = []
    for ip in ips[:5]:                 # cap at 5 to avoid huge messages
        data = await ip_info(ip)
        blocks.append(format_ip_block(ip, data))

    result = f"✅ *{_esc(domain)}*\n\n" + "\n".join(blocks)
    if len(result) > 4000:
        result = result[:4000] + "\n… (truncated)"

    try:
        await wait.edit_text(result, parse_mode="Markdown")
    except Exception:
        # Fallback: plain text if Markdown parsing fails
        await wait.edit_text(result)


# ------------------------------------------------------------------
# MAIN
# ------------------------------------------------------------------
async def main():
    global session
    session = aiohttp.ClientSession()
    print("Bot started. Send a domain in Telegram.")
    try:
        await dp.start_polling(bot)
    finally:
        await session.close()
        await bot.session.close()


if __name__ == "__main__":
    asyncio.run(main())
