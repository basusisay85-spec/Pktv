/* ================= ELEMENTS ================= */
const video        = document.getElementById("video");
const overlay      = document.getElementById("overlay");
const nowPlaying   = document.getElementById("nowPlaying");
const channelList  = document.getElementById("channelList");
const netStatus    = document.getElementById("netStatus");

/* ================= HLS ================= */
let hls = null;
function destroyHls() {
  if (hls) { hls.destroy(); hls = null; }
}

function playStream(ch, el) {
  destroyHls();
  document.querySelectorAll(".channel").forEach(c => c.classList.remove("active"));
  el.classList.add("active");

  nowPlaying.textContent = ch.name;
  overlay.textContent = "Loading…";
  overlay.classList.remove("hidden");

  const fail = (msg) => {
    overlay.textContent = "⚠ " + msg;
    overlay.classList.remove("hidden");
    nowPlaying.textContent = ch.name + " — error";
  };

  if (ch.type === "hls") {
    if (video.canPlayType("application/vnd.apple.mpegurl")) {
      // Safari / iOS — native HLS
      video.src = ch.src;
      video.play().catch(() => {});
      overlay.classList.add("hidden");
    } else if (window.Hls && Hls.isSupported()) {
      hls = new Hls({ lowLatencyMode: true, enableWorker: true });
      hls.loadSource(ch.src);
      hls.attachMedia(video);
      hls.on(Hls.Events.MANIFEST_PARSED, () => {
        overlay.classList.add("hidden");
        video.play().catch(() => {});
      });
      hls.on(Hls.Events.ERROR, (_e, data) => {
        if (data.fatal) {
          switch (data.type) {
            case Hls.ErrorTypes.NETWORK_ERROR: hls.startLoad(); break;
            case Hls.ErrorTypes.MEDIA_ERROR:   hls.recoverMediaError(); break;
            default: fail("stream unavailable"); destroyHls(); break;
          }
        }
      });
    } else {
      fail("HLS not supported in this browser");
    }
  } else {
    // mp4
    video.src = ch.src;
    video.play().then(() => overlay.classList.add("hidden"))
               .catch(() => fail("tap play to start"));
  }
}

/* ================= RENDER CHANNELS ================= */
CHANNELS.forEach(ch => {
  const div = document.createElement("div");
  div.className = "channel";
  div.innerHTML = `
    <div class="icon">${ch.icon}</div>
    <div class="info">
      <div class="cname">${ch.name}</div>
      <div class="cmeta">${ch.meta}</div>
    </div>`;
  div.addEventListener("click", () => playStream(ch, div));
  channelList.appendChild(div);
});

/* ================= NETWORK STATUS ================= */

    netStatus.textContent = "● Online";
    netStatus.style.background = "rgba(34,197,94,.15)";
    netStatus.style.color = "#22c55e";
    netStatus.style.borderColor = "rgba(34,197,94,.3)";
  } else {
    netStatus.textContent = "● Offline";
    netStatus.style.background = "rgba(239,68,68,.15)";
    netStatus.style.color = "#ef4444";
    netStatus.style.borderColor = "rgba(239,68,68,.3)";
  }
}

✅ What's inside

Part What it does
Splash Your original stadium animation, completed — ball flies, players run, title fades in, auto-hides after 4.2 s
Player <video> + hls.js with fatal-error recovery (network + media error types)
Native HLS fallback Works on Safari / iOS without hls.js
Channel grid Auto-rendered from the CHANNELS array, active state, loading/error overlays
Now Playing bar Blinking red dot + channel name
Online/Offline badge Updates via navigator.onLine events

🔧 To use your own streams

Edit only the CHANNELS array near the top of the <script>:

{ name: "Star Sports", meta: "Live · HLS", icon: "🏏", type: "hls",
  src: "https://your-stream-url.m3u8" },
  • type: "hls" → any .m3u8
  • type: "mp4" → any direct .mp4

⚠️ Note: browsers require HTTPS for mixed content — if you host this, use https:// stream URLs, or the request will be blocked.
